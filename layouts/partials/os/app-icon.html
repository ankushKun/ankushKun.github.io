{{/* Shared app icon renderer. Usage: {{ partial "os/app-icon.html" (dict "app" $app "context" "desktop") }} Contexts map to fixed visual sizes in CSS: desktop 64px, grid 48px, spotlight/notification 32px, chrome 18px. */}} {{- $app := .app -}} {{- $icon := .icon -}} {{- if and (not $icon) $app -}}{{- $icon = $app.icon -}}{{- end -}} {{- $context := default "desktop" .context -}} {{- $fit := "contain" -}} {{- with $icon -}}{{- with .fit -}}{{- $fit = . -}}{{- end -}}{{- end -}} {{- $iconType := "fallback" -}} {{- with $icon -}}{{- $iconType = .type -}}{{- end -}} {{- $classes := slice "app-icon" (printf "app-icon-%s" $context) (printf "app-icon-type-%s" $iconType) (printf "app-icon-fit-%s" $fit) -}} {{- with .class -}}{{- $classes = $classes | append . -}}{{- end -}} {{- $loading := "eager" -}} {{- if eq $context "desktop" -}}{{- $loading = "lazy" -}}{{- end -}} {{- if and $icon (eq $icon.type "image") -}} {{- else if and $icon (eq $icon.type "partial") -}} {{ partial $icon.partial $app }} {{- else -}} {{ partial "icons/desktop/document.html" $app }} {{- end -}}

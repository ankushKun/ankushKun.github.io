{{/* Shared app icon renderer. Usage: {{ partial "os/app-icon.html" (dict "app" $app "context" "desktop") }} Contexts map to fixed visual sizes in CSS: desktop 64px, grid 48px, spotlight/notification 32px, chrome 18px. */}} {{- $app := .app -}} {{- $icon := .icon -}} {{- if and (not $icon) $app -}}{{- $icon = $app.icon -}}{{- end -}} {{- $context := default "desktop" .context -}} {{- $fit := "contain" -}} {{- with $icon -}}{{- with .fit -}}{{- $fit = . -}}{{- end -}}{{- end -}} {{- $iconType := "fallback" -}} {{- with $icon -}}{{- $iconType = .type -}}{{- end -}} {{- $classes := slice "app-icon" (printf "app-icon-%s" $context) (printf "app-icon-type-%s" $iconType) (printf "app-icon-fit-%s" $fit) -}} {{- with .class -}}{{- $classes = $classes | append . -}}{{- end -}} {{/* Desktop icons are on screen at first paint. Every other context lives in a hidden template or a list that is cloned on demand, where an eager image would be fetched on every page view without ever being shown. */}} {{- $loading := "lazy" -}} {{- if eq $context "desktop" -}}{{- $loading = "eager" -}}{{- end -}} {{- $px := index (dict "desktop" 64 "grid" 48 "spotlight" 32 "notification" 32 "chrome" 18) $context | default 64 -}} {{- if and $icon (eq $icon.type "image") -}} {{- else if and $icon (eq $icon.type "partial") -}} {{- $art := partial $icon.partial $app -}} {{- /* The desktop icon keeps the artwork's own gradient ids. Every other copy (list toolbars, Spotlight data) gets them suffixed with the context, so the home page never holds two elements with the same id and each copy's url(#...) points at its own definitions. */ -}} {{- if ne $context "desktop" -}} {{- $sfx := printf "-%s" $context -}} {{- $art = $art | replaceRE `(\sid=")([^"]+)(")` (printf "${1}${2}%s${3}" $sfx) -}} {{- $art = $art | replaceRE `(url\(\s*['"]?#)([^'")\s]+)` (printf "${1}${2}%s" $sfx) -}} {{- $art = $art | replaceRE `((?:xlink:)?href="#)([^"]+)` (printf "${1}${2}%s" $sfx) -}} {{- $art = $art | safeHTML -}} {{- end -}} {{ $art }} {{- else -}} {{ partial "icons/desktop/document.html" $app }} {{- end -}}
